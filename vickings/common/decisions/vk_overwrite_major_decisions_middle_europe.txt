### Restore the Holy Roman Empire ###
restore_holy_roman_empire_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_dynasty_house.dds"
	}
	decision_group_type = major

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 12
		empire = 12
		hegemony = 0
	}

	desc = restore_holy_roman_empire_decision_desc
	selection_tooltip = restore_holy_roman_empire_decision_tooltip

	is_shown = {
		NOR = {	#Title existence conditions.
			exists = title:e_hre.holder	#Well, that'd be redundant.
		}
		OR = {	#Government conditions.
			government_has_mechanic = administrative
			government_has_flag = government_is_feudal
			government_has_flag = government_is_clan
		}
		AND = {	#Faith conditions.
			faith = {
				religion_tag = christianity_religion
			}
			trigger_if = {
				limit = { exists = faith.religious_head }
				this != faith.religious_head
			}
			trigger_else = {
				NOT = { faith.main_rite.head_of_rite ?= this }
			}
			#NAND = {	#No need for a competing empire if your faith controls the ERE.
			#	exists = title:e_byzantium.holder
			#	faith = title:e_byzantium.holder.faith
			#}
		}
		NOR = { #It doesn't make sense for these competing empires.
			is_roman_emperor_trigger = yes
			mpo_has_gok_mongol_empire_trigger = yes

			# CHANGED: added
			exists = title:e_austria.holder
			exists = title:e_germany.holder
		}
		highest_held_title_tier < tier_hegemony	# Don't destroy your hegemony.
		NOT = { has_global_variable = flag_restored_roman_empire } # Not if Rome already exists (prevent Rome from restoring HRE)
		is_landed_or_landless_administrative = yes
		exists = dynasty # Everyone should have a dynasty, but in case they don't...
	}

	is_valid = {
		#Standard requirements.
		top_liege = this
		custom_description = {
			text = restore_hre_mainline_rite_requirement_tt
			rite = faith.main_rite
		}
		trigger_if = {
			limit = {
				is_ai = no
			}
			prestige_level >= high_prestige_level
		}
		#Title ownership conditions.
		custom_tooltip = {
			text = decision_refound_hre_kingdom_list.tt
			OR = { #Gotta be King of a core Carolingian Kingdom.
				has_title = title:k_east_francia
				has_title = title:k_bavaria
				has_title = title:k_france
				has_title = title:k_italy
				has_title = title:k_lotharingia
				any_vassal = {
					OR = {
						has_title = title:k_east_francia
						has_title = title:k_bavaria
						has_title = title:k_france
						has_title = title:k_italy
						has_title = title:k_lotharingia
					}
				}
			}
		}
		trigger_if = {
			limit = {
				is_ai = no
			}
			custom_description = {	#And have some additional royal dignity for good measure.
				text = decision_refound_hre_own_three_kingdoms.tt
				any_held_title = {
					title_tier = kingdom
					count >= 3
				}
			}
		}
		sub_realm_size >= 120	#Plus your various kingdoms should be ruling something.
		# Must be 7 potential Electors
		custom_tooltip = {
			text = restore_holy_roman_empire_decision_trigger_de_jure_vassals.tt
			any_vassal = {
				count >= 7
				hre_valid_prince_elector_trigger = yes
				# prince-bishopric or duchy
				trigger_if = {
					limit = { government_has_flag = government_is_theocracy }
					any_held_title = {
						tier >= tier_county
						is_landless_type_title = no
						has_clerical_region = no
					}
				}
				trigger_else = {
					OR = {
						primary_title.tier >= tier_duchy
						has_character_flag = hre_kingdom_destroyed
					}
				}
			}
		}
		rule_title_creation_imperial_power_projection_title_creation_trigger = yes
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
		NOT = { has_trait = excommunicated }
	}

	effect = {
		hre_assign_dynamic_prince_electors_effect = yes
		save_scope_as = founder
		if = {
			limit = { exists = faith.religious_head }
			faith.religious_head = { save_scope_as = founder_religious_authority }
		}
		else_if = {
			limit = { exists = faith.main_rite.head_of_rite }
			faith.main_rite.head_of_rite = { save_scope_as = founder_religious_authority }
		}
		capital_province ?= { save_scope_as = ceremony_locale }
		if = {
			limit = {
				exists = scope:founder_religious_authority
				scope:founder_religious_authority = {
					is_available_adult = yes
					OR = {
						opinion = {
							target = root
							value >= high_positive_opinion
						}
						root = { has_strong_usable_hook = this }
						root = { has_weak_hook = this }
					}
				}
				NOT = { is_at_war_with = scope:founder_religious_authority }
			}
			scope:founder_religious_authority = {
				save_scope_as = hre_blessing_authority
				if = {
					limit = { is_playable_character = yes }
					capital_province ?= { save_scope_as = ceremony_locale }
				}
			}
		}
		if = {
			limit = { exists = scope:hre_blessing_authority }
			custom_tooltip = restore_hre_with_blessing_tt
		}
		else = {
			custom_tooltip = restore_hre_without_blessing_tt
		}
		show_as_tooltip = {
			restore_holy_roman_empire_decision_scripted_effect = yes
		}
		custom_tooltip = hre_elector_list_creation_tt
		gain_heroic_legend_seed_tooltip_effect = yes
		trigger_event = middle_europe_decisions.0015
		#Notify other players.
		every_player = {
			limit = {
				this != root
				is_within_diplo_range = { CHARACTER = root }
			}
			trigger_event = middle_europe_decisions.0016
		}
		if = {
			limit = {
				is_ai = no
				NOT = { exists = global_var:restore_holy_roman_empire_decision  }
			}
			set_global_variable = {
				name = restore_holy_roman_empire_decision
				value = root
			}
		}
	}

	cost = {
		gold = {
			value = 0
			if = {
				limit = {
					has_treasury = no
					is_ai = no
				}
				add = 500
			}
		}
		treasury = {
			value = 0
			if = {
				limit = {
					has_treasury = yes
					is_ai = no
				}
				add = 500
			}
		}
		piety = 200
	}

	ai_potential = {
		top_liege = this
		highest_held_title_tier >= tier_kingdom
	}

	ai_will_do = {
		base = 100

		modifier = { # We don't want a solo-France to form the HRE
			factor = 0
			NOR = {
				has_title = title:k_east_francia
				has_title = title:k_bavaria
				has_title = title:k_italy
				has_title = title:k_lotharingia
				any_vassal = {
					OR = {
						has_title = title:k_east_francia
						has_title = title:k_bavaria
						has_title = title:k_italy
						has_title = title:k_lotharingia
					}
				}
			}
		}
	}
}
