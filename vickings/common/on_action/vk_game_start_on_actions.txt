on_game_start = {
	on_actions = {
		vk_initialize_history_effects
	}
}

vk_initialize_history_effects = {
	effect = {
		auto_generate_vassals_at_game_start_effect = yes
		shatter_tribals_at_game_start_effect = yes
		vk_setup_tributaries_effect = yes
	}
}

# Like on_game_start, except it is called once the host (or player, in single player) exits the lobby. Good for anything where you need to know who the players are, or what the game rules are
on_game_start_after_lobby = {
	on_actions = {
		vk_initialize_history_rules_effects
	}
}

vk_initialize_history_rules_effects = {
	effect = {
		if = {
			limit = { has_game_rule = bd_occurrence_historical }
			set_global_variable = black_death_happened
		}
		### TGP SETUP ###
		if = {
			limit = { has_tgp_dlc_trigger = yes }

			### MANDALA

			#Starting Mandala Temple Complexes
			#Pagan
			province:9550 = {
				ruin_great_building = yes
			}
			#Angkor
			province:10816 = {
				ruin_great_building = yes
			}
			#Vijaya
			province:10827 = {
				ruin_great_building = yes
			}

			## 19th century Mandala Capitals
			#Ava
			province:9556 = {
				add_great_building = mandala_capital_03
				add_to_global_variable_list = { # Add to list so it can be found by the PoI
					name = mandala_poi_list
					target = this
				}
			}
			#Vien Chan
			province:10820 = {
