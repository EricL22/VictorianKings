# generate governors for large realms
auto_generate_vassals_at_game_start_effect = {
	every_ruler = {
		limit = {
			primary_title.tier > tier_duchy
			OR = {
				sub_realm_size > 20
				primary_title = title:k_funan	# fix for claimant war bug
			}
			OR = {
				is_independent_ruler = yes
				primary_title.tier > tier_kingdom
			}
			NOT = { primary_title = title:e_japan }
		}
		save_temporary_scope_as = ruler
		every_realm_de_jure_duchy = {
			limit = {
				NOT = { THIS = scope:ruler.capital_county.de_jure_liege }
				NOT = { is_target_in_global_variable_list = { name = personal_union target = THIS } }
			}
			create_character = {
				location = title_capital_county.title_province
				culture = scope:ruler.culture
				faith = scope:ruler.faith
				rite = scope:ruler.rite
				gender_female_chance = 5
				age = 32
				save_scope_as = backup_character
			}
			create_title_and_vassal_change = {
				type = granted
				save_scope_as = change
				add_claim_on_loss = no
			}
			every_de_jure_county = {
				limit = { holder = scope:ruler }
				change_title_holder = {
					holder = scope:backup_character
					change = scope:change
				}
			}
			if = {
				limit = { holder = scope:ruler }
				change_title_holder = {
					holder = scope:backup_character
					change = scope:change
				}
			}
			resolve_title_and_vassal_change = scope:change
			scope:backup_character = {
				if = {
					limit = { has_government = celestial_government }
					add_character_flag = passed_provincial_exam
					add_character_modifier = {
						modifier = tgp_passed_provincial_exam_modifier
					}
					add_character_flag = passed_metropolitan_exam
					add_character_modifier = {
						modifier = tgp_passed_metropolitan_exam_modifier
					}
					change_merit = 5000
				}
			}
		}
	}
}

# mark a personal union title, which will gain independence from parent realm on succession if holder changes
set_personal_union_effect = {
	add_to_global_variable_list = {
		name = personal_union
		target = $TITLE$
	}
}

# clear a personal union title, reverting back to default behavior on succession
clear_personal_union_effect = {
	remove_list_global_variable = {
		name = personal_union
		target = $TITLE$
	}
}

# The setup_tributaries_effect is invoked by on_game_start and should make calls to subsequent scripted effects.
vk_setup_tributaries_effect = {

		limit = { has_tgp_dlc_trigger = yes }
		if = {
			limit = { current_date < 1836.1.1 }
			tgp_pre_1836_tributary_setup_effect = yes
		}
		if = {
			limit = { current_date < 1914.7.28 }
			tgp_pre_1914_tributary_setup_effect = yes
		}
	}
	if = {
		limit = {
			current_date > 1812.7.5
		}
		base_post_1812_tributary_setup_effect = yes
	}
   	if = {
		limit = {
			current_date < 1861.7.1
		}
		base_pre_1861_tributary_setup_effect = yes
	}
   	if = {
		limit = {
			current_date = 1861.7.1
		}
		base_1861_tributary_setup_effect = yes
	}
   	if = {
		limit = {
			current_date < 1914.7.28
		}
		base_pre_1914_tributary_setup_effect = yes
	}
}
